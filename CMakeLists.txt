# Translations: every res/lang/<code>.lang (not template.lang) is built into bytopsy.exe as a
# resource of type LANG named after its code ("pt-BR" -> PT_BR). A new file needs a CMake re-run,
# which CONFIGURE_DEPENDS makes the next build do by itself.
file(GLOB BYTOPSY_LANG_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/res/lang/*.lang")
set(BYTOPSY_LANG_RC "// Generated by CMakeLists.txt from res/lang/*.lang\n")
set(BYTOPSY_LANG_DEPENDS "")
foreach(f ${BYTOPSY_LANG_FILES})
  get_filename_component(code "${f}" NAME_WE)
  string(TOLOWER "${code}" code_lower)
  if(NOT code_lower STREQUAL "template")
    string(TOUPPER "${code}" res_name)
    string(REPLACE "-" "_" res_name "${res_name}")
    string(APPEND BYTOPSY_LANG_RC "${res_name} LANG \"${f}\"\n")
    list(APPEND BYTOPSY_LANG_DEPENDS "${f}")
  endif()
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/lang_resources.rc" CONTENT "${BYTOPSY_LANG_RC}")
target_include_directories(bytopsy PRIVATE "${CMAKE_BINARY_DIR}")
# The resource script embeds these files; without this, editing them doesn't rebuild the exe.
set_source_files_properties(src/gui/bytopsy-gui.rc PROPERTIES
  OBJECT_DEPENDS "${CMAKE_SOURCE_DIR}/res/cleaners.ini;${CMAKE_SOURCE_DIR}/res/bytopsy.ico;${CMAKE_BINARY_DIR}/lang_resources.rc;${BYTOPSY_LANG_DEPENDS}")

endif()
